-- This does NOT delete any documents or user data.
drop policy if exists "Users manage their own documents" on documents;
drop policy if exists "Users manage their own conversations" on conversations;
drop policy if exists "Users manage their own chunks" on chunks;
drop policy if exists "Users manage messages in their own conversations" on messages;
